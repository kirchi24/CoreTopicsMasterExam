\section{Informatik}

\subsection{Relationale vs. multidimensionale Datenbanken}

\definition{\textbf{Datenbanken} (DB) sind strukturierte, persistente Sammlungen von Daten. Ein \textbf{Datenbankmanagementsystem} (DBMS) ist die Software zur Verwaltung dieser Datenbanken (z.\,B.\ PostgreSQL, Oracle, MySQL).}

Datenbanksysteme unterscheiden sich primär durch ihren Einsatzzweck: operative Systeme brauchen schnelle, kleine Transaktionen auf aktuellen Daten, analytische Systeme brauchen komplexe Abfragen über große Datenmengen.

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{1.8cm} L{7.7cm} L{5cm}}
    & \textbf{OLTP} (Online Transaction Processing) & \textbf{OLAP} (Online Analytical Processing) \\
    \midrule
    Zweck & operative Systeme: Banking, Webshops, Buchungen & analytische Auswertung: BI, Reporting, Data Mining \\
    Queries & klein, häufig, einzelne Records & komplex, große Datenmengen, Aggregationen \\
    Antwortzeit & Millisekunden & Sekunden bis Minuten \\
    Datenmodell & \textbf{normalisiert} (3NF/BCNF) -- vermeidet Update-Anomalien & \textbf{denormalisiert} -- optimiert Lesezugriffe, vermeidet JOINs \\
  \end{tabularx}
\end{table}

\definition{\textbf{Data Warehouse} (DWH): zentrale, integrierte, historisierte DB für analytische Abfragen. Daten fließen aus OLTP-Systemen via \textbf{ETL} (Extract, Transform, Load) ein.}

\textbf{Schema-Muster im DWH:} \textbf{Star-Schema} -- zentrale \textbf{Faktentabelle} (Measures: Umsatz, Menge) mit FKs zu \textbf{Dimensionstabellen} (Zeit, Produkt, Region); einfach, wenig JOINs. \textbf{Snowflake-Schema} -- Dimensionen zusätzlich normalisiert (z.\,B.\ Produkt\,$\to$\,Kategorie\,$\to$\,Hersteller); spart Speicher, mehr JOINs.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.65\textwidth]{assets/star_vs_snowflake.png}
\end{figure}

\textbf{Datenwürfel (OLAP-Kern):} $n$-dimensionale Struktur mit hierarchischen Dimensionen (Zeit: Jahr\,$\to$\,Quartal\,$\to$\,Tag). Operationen:
\begin{itemize}[leftmargin=1.5em,noitemsep]
    \item \textbf{Slice} -- eine Dimension fixieren (z.\,B.\ nur 2023)
    \item \textbf{Dice} -- Teilwürfel auswählen (z.\,B.\ 2023 \& Europa)
    \item \textbf{Drill-Down} -- zu Details navigieren \quad / \quad \textbf{Roll-Up} -- zu Aggregaten aufsteigen
\end{itemize}

\textbf{Data Lake:} speichert Rohdaten beliebigen Formats (strukturiert/semi/unstrukturiert); flexibler als DWH, erfordert aber starke Governance -- sonst Risiko der \emph{Data Swamp}.

\subsection{Entity-Relationship-Modell}

\definition{Das \textbf{Entity-Relationship-Modell} (ERM) ist ein konzeptionelles Modellierungsverfahren für die \textit{Miniwelt}. Kernkomponenten: \textbf{Entities} (Objekttypen, z.\,B.\ Person), \textbf{Attribute} (Eigenschaften), \textbf{Relationships} (Beziehungen zwischen Entitäten).}

\textbf{Modellierungskaskade:} konzeptionell (ER-Modell, fachliche Logik) $\to$ logisch (relationales Schema: Tabellen, Spalten, Schlüssel) $\to$ physisch (Speicherung im DBMS).

\textbf{Kardinalitäten} beschreiben, wie viele Entitäten je Seite beteiligt sind:
\begin{itemize}[leftmargin=1.5em,noitemsep]
    \item \textbf{1:1} (Person--Reisepass): je höchstens ein Partner auf jeder Seite
    \item \textbf{1:N} (Abteilung--Mitarbeiter): eine Abteilung hat viele Mitarbeiter, jeder Mitarbeiter genau eine Abteilung
    \item \textbf{N:M} (Student--Kurs): viele-zu-viele Zuordnung
\end{itemize}
\textbf{Optionalitäten} (Teilnahmebedingungen) ergänzen dies: optional (0..N) oder verpflichtend (1..N). Graphisch: einfacher Endpunkt = 1, Krähenfuß = N.

\textbf{Schwache Entitäten} besitzen keinen eigenen vollständigen Schlüssel, sondern werden über eine identifizierende Beziehung zu einer starken Entität bestimmt -- z.\,B.\ ist eine Bestellposition nur durch (Bestellnummer, Positionsnummer) eindeutig.

\textbf{Relationales Modell:} \textbf{Relation} = Tabelle, \textbf{Tupel} = Zeile, \textbf{Attribut} = Spalte, \textbf{Domäne} = zulässiger Wertebereich eines Attributs.

\textbf{Transformation ER $\to$ relational:} Jeder Entity-Typ wird zur Tabelle, Attribute zu Spalten. Bei Relationships: 1:N $\to$ \textbf{Fremdschlüssel} auf der N-Seite; N:M $\to$ separate \textbf{Verbindungstabelle} mit FKs zu beiden Seiten; 1:1 $\to$ FK oder Zusammenführung der Relationen.

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{2.5cm} L{4cm} L{8.5cm}}
    \textbf{Schlüsseltyp} & \textbf{Eigenschaft} & \textbf{Beschreibung und Verwendung}                                                \\
    \midrule
    Superkey              & Eindeutig            & Attributkombination, die Tupel eindeutig identifiziert (z.\,B.\ \{PersNr, Name\})   \\
    Candidate Key         & Minimal              & Minimaler Superkey; keine überflüssigen Attribute; mehrere möglich                  \\
    Primärschlüssel       & Gewählt              & Ein Candidate Key, ausgewählt zur Identifikation; NOT NULL, UNIQUE                  \\
    Fremdschlüssel        & Verweis              & Attribut, das einen PK einer anderen Tabelle referenziert                           \\
    Surrogate Key         & Künstlich            & Künstlich erzeugt (SERIAL, AUTO\_INCREMENT) für Eindeutigkeit                       \\
  \end{tabularx}
\end{table}

\textbf{Integritätsbedingungen:} \textbf{Entitätsintegrität} -- PK eindeutig und NOT NULL. \textbf{Referentielle Integrität} -- jeder FK ist NULL oder verweist auf existierenden PK. \textbf{Domänenintegrität} -- Attributwerte liegen innerhalb ihrer Domäne.

\subsection{SQL als Data Query Language}

\definition{\textbf{SQL} (Structured Query Language) ist die standardisierte Abfragesprache für relationale DBs, unterteilt in \textbf{DQL} (SELECT), \textbf{DML} (INSERT/UPDATE/DELETE), \textbf{DDL} (CREATE/ALTER/DROP), \textbf{DCL} (GRANT/REVOKE) und \textbf{TCL} (COMMIT/ROLLBACK).}

\textbf{Logische Ausführungsreihenfolge} (nicht die Schreibreihenfolge!): \texttt{FROM} $\to$ \texttt{WHERE} $\to$ \texttt{GROUP BY} $\to$ \texttt{HAVING} $\to$ \texttt{SELECT} $\to$ \texttt{ORDER BY} $\to$ \texttt{LIMIT/OFFSET}. Häufiger Fehler: Aggregatfunktionen in \texttt{WHERE} -- funktioniert nicht, da \texttt{WHERE} vor \texttt{GROUP BY} evaluiert wird; dafür ist \texttt{HAVING} (Filterung auf Gruppen) zuständig.

\definition{\textbf{NULL-Semantik:} NULL bedeutet ``unbekannt'', nicht ``leer''. \texttt{wert = NULL} ergibt \texttt{UNKNOWN} (dreiwertige Logik), nicht \texttt{FALSE} -- daher \texttt{IS NULL} / \texttt{IS NOT NULL} verwenden.}

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{4.0cm} L{4.5cm} L{6.5cm}}
    \textbf{Operator} & \textbf{Beispiel}               & \textbf{Beschreibung}                    \\
    \midrule
    BETWEEN                    & preis BETWEEN 10 AND 100        & Wertebereich (inklusive Grenzen)         \\
    IN                         & stadt IN ('Wien','Graz','Linz') & Wert muss in Menge enthalten sein        \\
    LIKE                       & name LIKE 'A\%'                 & Mustererkennung; \% beliebig, \_ einzeln \\
    IS NULL                    & alter IS NULL                   & Prüft auf NULL (nicht mit =)             \\
    COUNT(*) / COUNT(spalte)   & COUNT(*), COUNT(email)          & alle Zeilen / nur Nicht-NULL-Werte       \\
    SUM/AVG/ MIN/MAX            & SUM(umsatz), AVG(gewinn)        & Summe / Mittelwert / Extremwerte         \\
    COALESCE                   & COALESCE(bonus, 0)              & Erster Nicht-NULL-Wert                   \\
  \end{tabularx}
\end{table}

\textbf{JOINs:} \texttt{INNER JOIN} -- nur Matches beider Tabellen. \texttt{LEFT/RIGHT JOIN} -- alle Zeilen einer Seite, fehlende Matches mit NULL aufgefüllt. \texttt{FULL OUTER JOIN} -- alle Zeilen beider Tabellen. \texttt{CROSS JOIN} -- kartesisches Produkt aller Kombinationen.

\textbf{Beispiel} -- Tabellen \texttt{kunden(id, name)} und \texttt{bestellungen(id, kunde\_id, betrag)}:

\begin{verbatim}
-- INNER JOIN: Name + Betrag je Bestellung
SELECT k.name, b.betrag
FROM kunden k
JOIN bestellungen b ON b.kunde_id = k.id;

-- GROUP BY + AVG + HAVING: Kunden mit Schnitt > 100
SELECT k.name, AVG(b.betrag) AS schnitt
FROM kunden k
LEFT JOIN bestellungen b ON b.kunde_id = k.id
GROUP BY k.name
HAVING AVG(b.betrag) > 100;
\end{verbatim}
\texttt{LEFT JOIN} stellt hier sicher, dass auch Kunden ohne Bestellung in der Gruppierung erscheinen (mit \texttt{NULL}-Betrag); \texttt{HAVING} filtert erst \emph{nach} der Aggregation, \texttt{WHERE} könnte \texttt{AVG(b.betrag)} nicht referenzieren.

\textbf{Weitere Konzepte:} \texttt{DISTINCT} entfernt Duplikate; \textbf{Subqueries} sind verschachtelte SELECTs in WHERE/FROM/SELECT; \textbf{Views} sind virtuelle Tabellen auf Basis von Queries; \textbf{Indizes} beschleunigen Suchzugriffe.

\subsection{Normalformen, Integritätsbedingungen und Transaktionen}

\definition{\textbf{Normalisierung}: systematische Umstrukturierung zur Reduktion von Redundanz und Anomalien. \textbf{Insert-Anomalie} -- Entität nicht einfügbar ohne irrelevante Zusatzdaten. \textbf{Update-Anomalie} -- Information muss an mehreren Stellen synchron gepflegt werden. \textbf{Delete-Anomalie} -- Löschen verursacht ungewollten Datenverlust.}

\textbf{Durchgängiges Beispiel} -- Tabelle \texttt{Belegung(StudentID, Hobbys, KursID, KursName, Note)}, PK = (StudentID, KursID):

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{1.6cm} L{12.4cm}}
    \textbf{1NF} & Alle Attributwerte atomar, keine Wiederholungsgruppen. \textit{Verletzung:} ``Hobbys'' = ``Lesen, Schwimmen'' (mehrere Werte in einer Zelle) $\Rightarrow$ eigene Zeile/Tabelle pro Hobby. \\
    \textbf{2NF} & 1NF + keine \textbf{partiellen Abhängigkeiten}: bei zusammengesetztem PK darf kein Attribut nur von einem \emph{Teil} des PK abhängen. \textit{Verletzung:} KursName hängt nur von KursID ab, nicht vom vollen PK (StudentID, KursID) $\Rightarrow$ KursName in eigene Tabelle \texttt{Kurs(KursID, KursName)} auslagern. \\
    \textbf{3NF} & 2NF + keine \textbf{transitiven Abhängigkeiten}: ein Nicht-Schlüssel-Attribut darf nicht von einem \emph{anderen} Nicht-Schlüssel-Attribut abhängen (PK $\to$ A $\to$ B statt direkt PK $\to$ B). \textit{Verletzung:} in \texttt{Kurs(KursID, KursName, DozentID, DozentName)} hängt DozentName nicht direkt von KursID ab, sondern über DozentID $\Rightarrow$ \texttt{Dozent(DozentID, DozentName)} auslagern. \\
  \end{tabularx}
\end{table}
\textit{Merkhilfe:} 2NF betrifft die Abhängigkeit \emph{vom (Teil-)Schlüssel}, 3NF betrifft Abhängigkeiten \emph{zwischen Nicht-Schlüssel-Attributen}.

In der Praxis ist \textbf{3NF} der gängige Kompromiss zwischen Redundanzreduktion und Query-Performance; höhere Normalformen (BCNF, 4NF, 5NF) führen oft zu stark fragmentierten Schemas mit vielen JOINs.

\textbf{Integritätsbedingungen:} \texttt{PRIMARY KEY} (eindeutig, NOT NULL) $\cdot$ \texttt{FOREIGN KEY} (referentielle Integrität) $\cdot$ \texttt{UNIQUE} (eindeutig, NULL mehrfach erlaubt) $\cdot$ \texttt{NOT NULL} (Pflichtfeld) $\cdot$ \texttt{CHECK} (Bedingungsvalidierung, z.\,B.\ \texttt{Preis > 0}) $\cdot$ \texttt{DEFAULT} (Standardwert).

\textbf{ACID:} \textbf{Atomicity} -- unteilbar, ganz oder gar nicht. \textbf{Consistency} -- DB bleibt in gültigem Zustand. \textbf{Isolation} -- parallele Transaktionen beeinflussen sich nicht. \textbf{Durability} -- committete Daten überleben Systemausfälle.

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{3.2cm} L{9.8cm}}
    \textbf{Isolationslevel} & \textbf{Verhindert / erlaubt} \\
    \midrule
    Read Uncommitted & erlaubt \textbf{Dirty Read}: T1 liest einen von T2 geänderten, aber noch nicht committeten Wert; macht T2 ein Rollback, hat T1 mit Daten gearbeitet, die nie existierten. \\
    Read Committed & verhindert Dirty Reads, erlaubt \textbf{Non-repeatable Read}: T1 liest eine Zeile zweimal und bekommt unterschiedliche Werte, weil T2 dazwischen committet hat. \\
    Repeatable Read & verhindert Non-repeatable Reads, erlaubt \textbf{Phantom Read}: T1 wiederholt eine Bereichsabfrage und bekommt zusätzliche/fehlende Zeilen, weil T2 dazwischen neue Zeilen eingefügt/gelöscht hat. \\
    Serializable & verhindert alle drei Anomalien (serieller Zugriff, oft langsam). \\
  \end{tabularx}
\end{table}

\subsection{Datenschutz und Datensicherheit}

\textbf{Begriffsabgrenzung}

\textbf{Datenschutz} (\textit{data privacy}) und \textbf{Datensicherheit} (\textit{data security}) werden oft synonym verwendet, meinen aber Unterschiedliches:

\begin{itemize}[leftmargin=1.5em,noitemsep]
  \item \textbf{Datenschutz:} rechtliches/organisatorisches Konzept zum Schutz personenbezogener Daten vor unbefugter Verarbeitung. Kernfrage: \emph{Darf} man diese Daten erheben?
  \item \textbf{Datensicherheit:} technisches/organisatorisches Konzept zum Schutz aller Daten (unabhängig vom Personenbezug) vor Verlust, Manipulation, unbefugtem Zugriff. Kernfrage: \emph{Wie} sichert man die Daten technisch?
\end{itemize}

\textbf{Rechtlicher Rahmen: DSGVO}

Die \textbf{Datenschutz-Grundverordnung} (DSGVO, seit Mai 2018) ist die europäische Rechtsgrundlage für personenbezogene Daten -- gültig für alle Unternehmen, die Daten von EU-Bürgern verarbeiten, unabhängig vom Firmensitz.

\textbf{Grundsätze:} Rechtmäßigkeit \& Transparenz (Verarbeitung braucht Rechtsgrundlage: Einwilligung, Vertrag, gesetzliche Pflicht, berechtigtes Interesse) $\cdot$ Zweckbindung (nur für den festgelegten Zweck) $\cdot$ Datenminimierung (nur erforderliche Daten) $\cdot$ Richtigkeit $\cdot$ Speicherbegrenzung $\cdot$ Integrität \& Vertraulichkeit $\cdot$ Rechenschaftspflicht (Nachweispflicht des Verantwortlichen).

\textbf{Betroffenenrechte:} Auskunft, Berichtigung, Löschung (``Recht auf Vergessenwerden''), Datenportabilität, Widerspruch gegen automatisierte Entscheidungen.

\textbf{Sanktionen:} bis zu \textbf{20\,Mio.\ €} oder \textbf{4\,\% des weltweiten Jahresumsatzes}.

\textbf{CIA-Modell der Informationssicherheit}

\begin{itemize}[leftmargin=1.5em,noitemsep]
  \item \textbf{Confidentiality} (Vertraulichkeit): kein unbefugter Zugriff. \textit{Maßnahmen:} Verschlüsselung, Zugriffskontrolle
  \item \textbf{Integrity} (Integrität): keine unbefugte Veränderung. \textit{Maßnahmen:} Hashwerte, digitale Signaturen
  \item \textbf{Availability} (Verfügbarkeit): Zugriff jederzeit für Autorisierte. \textit{Maßnahmen:} Redundanz, Backups, DDoS-Schutz
\end{itemize}

\textbf{Technische und organisatorische Maßnahmen (TOMs)}

Die DSGVO verlangt ``Privacy by Design'' und angemessene TOMs, u.\,a.:
Pseudonymisierung/Anonymisierung $\cdot$ Verschlüsselung (AES-256 für Daten, TLS\,1.3 für Übertragung) $\cdot$ Role-Based Access Control (RBAC) $\cdot$ Backups nach 3-2-1-Regel (3 Kopien, 2 Medientypen, 1 Off-site) $\cdot$ Logging \& Auditing $\cdot$ regelmäßige Mitarbeiterschulungen.

\textbf{Datenschutz im ML/Data-Science-Kontext}

\begin{itemize}[leftmargin=1.5em,noitemsep]
  \item \textbf{Datenvergiftung:} unvorsichtiger Umgang mit Trainingsdaten kann zu Datenlecks führen
  \item \textbf{Model Inversion:} ML-Modelle können Trainingsdaten teilweise rekonstruieren
  \item \textbf{Fairness \& Bias:} diskriminierende Modelle verletzen die DSGVO
  \item \textbf{Datentransfers:} Übermittlung von EU-Daten in Drittländer oft unzulässig
\end{itemize}

\subsection{Python: Listen, Tupel, Mengen und Dictionaries}

\definition{\textbf{Listen} (\texttt{list}), \textbf{Tupel} (\texttt{tuple}), \textbf{Mengen} (\texttt{set}) und \textbf{Dictionaries} (\texttt{dict}) sind fundamentale Python-Datenstrukturen für Sammlungen. Sie unterscheiden sich in Veränderbarkeit, Ordnung, Eindeutigkeit und Verwendbarkeit als Dictionary-Keys. Alle vier sind iterierbar, unterstützen \texttt{in} sowie \texttt{len()} (\texttt{min(), max(), sum()} nur bei list/tuple/set bzw. auf \texttt{dict.values()}).}

\textbf{Liste:} \textbf{geordnet}, \textbf{veränderlich} (mutable), Duplikate erlaubt. Index-Zugriff $\mathcal{O}(1)$, Einfügen an beliebiger Position $\mathcal{O}(n)$ (Verschiebung nachfolgender Elemente). Einsatz: dynamische Sammlungen, wachsende Ergebnisse, Puffer, Warteschlangen.

\textbf{Tupel:} konzeptuell wie Liste, aber \textbf{unveränderlich} (immutable) -- keine \texttt{append()}/\texttt{remove()}. Dieser Constraint bringt zwei Vorteile: hashbar (also als Dictionary-Key nutzbar) und speichereffizienter, da kein Größen-Overhead vorgehalten wird. Mit \texttt{typing.NamedTuple} lassen sich benannte Felder definieren. Einsatz: Datenbankdatensätze, Funktionsrückgaben (Tuple Unpacking), Konfigurationsparameter.

\textbf{Menge (set):} ungeordnet, nur eindeutige Werte, intern als Hash-Tabelle. Mitgliedschaftstest, Einfügen, Löschen $\mathcal{O}(1)$ im Durchschnitt -- entscheidender Vorteil gegenüber der Liste ($\mathcal{O}(n)$) bei Existenztests in großen Kollektionen. Mengenoperationen: Vereinigung (\texttt{|}), Schnitt (\texttt{\&}), Differenz (\texttt{-}), symmetrische Differenz (\texttt{\textasciicircum}). \texttt{frozenset} ist die unveränderliche, hashbare Variante. Einsatz: Eindeutigkeitstests, Deduplizierung, Mengenlogik.

\textbf{Dictionary (dict):} Sammlung von \textbf{Key-Value-Paaren}, intern ebenfalls als Hash-Tabelle -- Keys müssen daher hashbar sein (z.\,B.\ \texttt{str}, \texttt{int}, \texttt{tuple}, \emph{nicht} \texttt{list}/\texttt{set}). Seit Python 3.7 \textbf{einfügereihenfolge-erhaltend}, aber nicht sortiert. Zugriff, Einfügen, Löschen über Key $\mathcal{O}(1)$ im Durchschnitt; Werte können beliebige (auch mutable) Objekte sein. \texttt{dict.get(key, default)} vermeidet \texttt{KeyError}; \texttt{collections.defaultdict} liefert automatisch einen Default-Wert für fehlende Keys. Einsatz: Lookup-Tabellen, Zählen/Gruppieren (z.\,B.\ Wort-Frequenzen), strukturierte Records ohne feste Klasse.

\begin{table}[H]
  \centering
  \begin{tabular}{l c c c c}
    \textbf{Eigenschaft} & \textbf{list}    & \textbf{tuple}   & \textbf{set}     & \textbf{dict}              \\
    \midrule
    Geordnet             & \checkmark       & \checkmark       & --               & \checkmark (seit 3.7)      \\
    Veränderlich         & \checkmark       & --               & \checkmark       & \checkmark                 \\
    Duplikate erlaubt    & \checkmark       & \checkmark       & --               & -- (bei Keys)              \\
    Zugriff/Suche        & $\mathcal{O}(n)$ & $\mathcal{O}(n)$ & $\mathcal{O}(1)$ & $\mathcal{O}(1)$ (per Key) \\
    Hashbar (als Key)    & --               & \checkmark       & --               & --                         \\
    Mengenoperationen    & --               & --               & \checkmark       & --                         \\
  \end{tabular}
\end{table}

\textbf{Wahl der Struktur:} Indexzugriff \& Veränderbarkeit nötig $\Rightarrow$ Liste. Unveränderlicher Key oder mehrere Rückgabewerte $\Rightarrow$ Tupel. Schneller Existenztest oder Mengenlogik $\Rightarrow$ Set. Zuordnung von Werten zu eindeutigen Schlüsseln (Lookup, Zählen) $\Rightarrow$ Dict.

\subsection{Python: Kopien vs. Verweise, Shallow \& Deep Copies}

\definition{In Python ist jede Variable ein \textbf{Name}, der auf ein Objekt im Speicher zeigt. \texttt{b = a} kopiert nicht das Objekt -- beide Namen zeigen auf \textbf{dasselbe Objekt} (Referenzsemantik).}

\texttt{is} prüft Identität (gleiche Speicheradresse), \texttt{==} prüft Gleichheit (gleicher Wert). Bei unveränderlichen Typen oft unsichtbar (Python cached kleine Integers/kurze Strings); bei veränderlichen Typen kritisch: zwei Listen können gleich (\texttt{==}), aber unterschiedliche Objekte (\texttt{is not}) sein.

\textbf{Unveränderliche Typen} (\texttt{int, float, str, tuple, frozenset}): scheinbare Änderungen erzeugen neue Objekte; ein Tupel selbst lässt sich nicht modifizieren. \textbf{Veränderliche Typen} (\texttt{list, dict, set}): Elemente können in-place verändert werden -- mit der Konsequenz, dass mehrere Variablen, die auf dieselbe Liste zeigen, sich Mutationen teilen:

\texttt{a = [1, 2, 3]; b = a; b.append(4)} \quad $\Rightarrow$ \quad \texttt{a} ist jetzt \texttt{[1, 2, 3, 4]} (kein Tippfehler -- \texttt{a} und \texttt{b} sind derselbe Name für dasselbe Objekt, \texttt{a is b} liefert \texttt{True}).

\textbf{Shallow Copy:} neuer Container (äußerste Ebene), Elemente bleiben geteilt -- bei verschachtelten Strukturen oft unzureichend. Wege: \texttt{list(orig)}, \texttt{orig[:]}, \texttt{copy.copy(orig)}, \texttt{orig.copy()}.

\textbf{Deep Copy:} vollständig unabhängiges Objekt inkl. aller rekursiv erreichbaren Unterobjekte, keine geteilten mutable Objekte. Standard: \texttt{copy.deepcopy(orig)} (behandelt auch zyklische Referenzen via Memo-Dictionary). Deutlich teurer -- in Performance-kritischem Code vermeiden, stattdessen unveränderliche Strukturen oder explizite Rekonstruktion nutzen.

\textbf{Beispiel} -- verschachtelte Liste \texttt{orig = [[1, 2], [3, 4]]}:
\begin{center}
\texttt{shallow = copy.copy(orig)} \quad\quad \texttt{deep = copy.deepcopy(orig)} \quad\quad \texttt{orig[0].append(99)}
\end{center}
Ergebnis: \texttt{orig} und \texttt{shallow} sind beide \texttt{[[1, 2, 99], [3, 4]]} -- die \emph{äußere} Liste wurde bei \texttt{shallow} zwar kopiert, die \emph{innere} Liste \texttt{orig[0]} aber nicht (\texttt{orig[0] is shallow[0]} $\to$ \texttt{True}). \texttt{deep} bleibt unverändert bei \texttt{[[1, 2], [3, 4]]}, da \texttt{deepcopy} auch die innere Liste dupliziert (\texttt{orig[0] is deep[0]} $\to$ \texttt{False}).

\textbf{Funktionsparameter} folgen \textbf{Pass-by-Object-Reference}: die Funktion erhält eine Referenz; Mutation eines veränderlichen Objekts wirkt nach außen, eine Neuzuweisung (\texttt{param = param + 1}) nicht.

\textbf{Default-Argument-Falle:} Ein mutables Default-Argument wird nur \emph{einmal} bei Funktionsdefinition erstellt und über alle Aufrufe geteilt:

\texttt{def f(lst=[]):\ lst.append(1);\ return lst}

Drei aufeinanderfolgende Aufrufe \texttt{r1 = f(); r2 = f(); r3 = f()} liefern \texttt{r1 = [1]}, \texttt{r2 = [1, 1]}, \texttt{r3 = [1, 1, 1]} -- nicht dreimal \texttt{[1]}, wie man naiv erwarten würde. Grund: \texttt{lst=[]} wird nur \emph{einmal} beim Definieren von \texttt{f} ausgewertet und im Funktionsobjekt gespeichert; jeder Aufruf ohne explizites Argument mutiert dieselbe Liste weiter (\texttt{r1 is r2 is r3} $\to$ \texttt{True}, alle drei Namen zeigen am Ende auf dasselbe, dreimal veränderte Objekt).

\textbf{Korrekt:} \texttt{def f(lst=None):\ if lst is None:\ lst = []} -- so wird bei jedem Aufruf ohne Argument eine \emph{neue} leere Liste erzeugt.

\subsection{Python: Schritte des Data-Preprocessings (scikit-learn)}

\definition{\textbf{Data-Preprocessing} bereitet Rohdaten systematisch für ML-Algorithmen vor. Viele Verfahren (SVM, k-NN, neuronale Netze) sind empfindlich gegenüber Skalierung, fehlenden Werten oder falscher Kodierung.}

scikit-learn strukturiert Preprocessing über das \textbf{Pipeline-Konzept}: jeder Schritt hat \texttt{fit(X)} (auf Trainingsdaten lernen) und \texttt{transform(X)} (anwenden); mehrere Schritte werden via \texttt{Pipeline} verkettet. \textbf{Kritisch:} \texttt{fit} nur auf Trainingsdaten -- sonst \textbf{Data Leakage} (Test-Set beeinflusst Preprocessing-Parameter, unrealistisch optimistische Metriken).

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{2.6cm} L{4.6cm} L{6.8cm}}
    \textbf{Schritt} & \textbf{Werkzeuge} & \textbf{Beschreibung} \\
    \midrule
    1. Imputation & \texttt{SimpleImputer}, \texttt{KNNImputer} & Strategien: \texttt{mean} (Durchschnitt), \texttt{median} (ausreißerrobust), \texttt{most\_frequent} (Modus, kategorisch), \texttt{constant}; KNNImputer nutzt Nachbarwerte \\
    2. Skalierung & \texttt{StandardScaler}, \texttt{MinMaxScaler}, \texttt{RobustScaler} & $z=(x-\mu)/\sigma$ (Standard); $[0,1]$-Normierung (ausreißerempfindlich); IQR-basiert (ausreißerrobust) \\
    3. Kodierung & \texttt{OrdinalEncoder}, \texttt{OneHotEncoder} & Ordinal: Ganzzahlen (nur bei echter Ordnung); OneHot: binäre Dummy-Spalten je Kategorie, vermeidet implizite Rangfolge \\
    4. Feature Engineering & \texttt{PolynomialFeatures}, \texttt{PowerTransformer} & Interaktionsterme/Potenzen ($x_1^2, x_1x_2,\ldots$); Yeo-Johnson normalisiert schiefe Verteilungen \\
    5. Feature-Selektion & \texttt{VarianceThreshold}, \texttt{SelectKBest}, \texttt{SelectFromModel} & entfernt (quasi-)konstante Features; Top-$k$ nach Teststatistik; Auswahl via Modell-Importances \\
  \end{tabularx}
\end{table}

\textbf{Class Imbalance:} bei stark unbalancierten Klassen (z.\,B.\ 99\,\%/1\,\%) klassifiziert das Modell sonst trivial die Mehrheitsklasse. Abhilfen: SMOTE (synthetisches Oversampling der Minoritätsklasse) oder \texttt{class\_weight='balanced'}.

\textbf{ColumnTransformer:} wendet unterschiedliche Pipelines parallel auf verschiedene Feature-Gruppen an -- z.\,B.\ numerische Features mit StandardScaler, kategorische mit OneHotEncoder.

\subsection{NumPy -- Einsatzmöglichkeiten und Vorteile gegenüber \glqq plain\grqq{} Python}

\definition{\textbf{NumPy} (Numerical Python) stellt homogene, multidimensionale Arrays (in Python (\texttt{numpy.ndarray})) und mathematische Funktionen bereit -- die Basis für SciPy, pandas, scikit-learn, PyTorch, TensorFlow.}

\textbf{Performance-Vorteil} (typisch 50--100$\times$ schneller als plain Python) resultiert aus: \textbf{homogenem, kompaktem Speicher} (alle Elemente gleicher Typ, z.\,B.\ \texttt{float64}, zusammenhängend gespeichert -- kein Objekt-Overhead wie bei Listen); \textbf{Vektorisierung} (keine expliziten Python-Schleifen, nutzt SIMD-Instruktionen); \textbf{Broadcasting} (Kombination unterschiedlicher Shapes ohne Datenkopie); \textbf{C/Fortran-Kernroutinen} für performancekritische Teile.

\textbf{Kernkonzepte:} \texttt{dtype} legt einheitlichen Datentyp fest (\texttt{float64, int32, bool,} \ldots); \texttt{shape} beschreibt Dimensionen (z.\,B.\ \texttt{(100,28,28)} für 100 Bilder à 28$\times$28). Viele Operationen (Slicing, Transpose, Reshape) liefern \textbf{Views} -- geteilter Speicherpuffer mit dem Original (prüfbar via \texttt{arr.base}).

\textbf{Broadcasting-Regeln:} Shapes werden \textbf{von rechts} ausgerichtet; zwei Dimensionen sind kompatibel, wenn sie \textbf{gleich} sind oder \textbf{eine von beiden 1} ist (die ``1''-Dimension wird gedanklich wiederholt, ohne zu kopieren). Fehlt eine Dimension, wird sie links als 1 ergänzt. Stimmt an einer Position keine der beiden Bedingungen, wirft NumPy einen Fehler.

\textbf{Beispiel} -- Spaltenvektor (3,1) + Zeilenvektor (3,) ergibt eine $3\times3$-Matrix (jede Dimension der Größe 1 wird auf die Größe der anderen Seite ``aufgespannt''):
\[
\underbrace{\begin{pmatrix}10\\20\\30\end{pmatrix}}_{(3,1)} +\ \underbrace{\begin{pmatrix}1 & 2 & 3\end{pmatrix}}_{(3,)\,\to\,(1,3)} =\ \underbrace{\begin{pmatrix}11 & 12 & 13\\21 & 22 & 23\\31 & 32 & 33\end{pmatrix}}_{(3,3)}
\]
\textit{Gegenbeispiel (Fehler):} (3,4) + (3,) schlägt fehl, da die letzten Dimensionen $4$ und $3$ weder gleich noch $1$ sind; (3,4) + (4,) funktioniert dagegen (die 4 passt direkt).

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{2.8cm} L{11.2cm}}
    \textbf{Kategorie} & \textbf{Funktionen} \\
    \midrule
    Erstellung & \texttt{np.zeros, np.ones, np.arange, np.linspace, np.random.rand} \\
    Indexing & \texttt{a[2:5]} (Slicing), \texttt{a[::2]} (jedes 2.), \texttt{a[a > 0]} (Boolean Mask) \\
    Aggregation & \texttt{sum(), mean(), std(), max(axis=0)} \\
    Lin.\ Algebra & \texttt{np.dot, a @ b} (Matrixmultiplikation), \texttt{np.linalg.inv, eig, solve} \\
    Ufuncs & \texttt{np.exp, log, sqrt, sin} -- elementweise vektorisiert \\
  \end{tabularx}
\end{table}

\textbf{Einsatzgebiete:} Lineare Algebra (Gleichungssysteme, Eigenwertzerlegung, FFT); Machine Learning (Feature-Matrizen, kompatibel mit scikit-learn, nahtlose PyTorch-Konvertierung); Bilddaten ($(H,W,3)$-Arrays, Normalisierung, Cropping); Simulation (ODE-Schritte, Interpolation ohne Python-Schleifen).

\textbf{Grenzen:} nur CPU (für GPU: CuPy, PyTorch, TensorFlow); keine automatische Differentiation (dafür PyTorch oder JAX).

\subsection{R: Vektoren, Matrizen, Arrays, DataFrames, Listen}

\definition{\textbf{R} ist eine funktionale, objektorientierte Sprache spezialisiert auf Statistik und Data Science. Grundphilosophie: \textbf{Alles ist ein Vektor} -- auch ein einzelner Wert ist ein Vektor der Länge 1. Die Grunddatenstrukturen unterscheiden sich in Homogenität, Dimensionalität und Verwendungszweck.}

\textbf{Atomare Vektoren} sind die Bausteine: homogen, Typen \texttt{numeric, integer, logical, character, complex, raw}. Zentrale Eigenschaft ist \textbf{Vektorisierung} -- \texttt{x + y} oder \texttt{sin(x)} arbeiten elementweise ohne explizite Schleife. \textbf{Recycling:} ein kürzerer Vektor wird automatisch wiederholt, bis die Länge passt. Einsatz: numerische Berechnungen, Zeitreihen, Messreihen.

\textbf{Matrizen} sind 2D-Verallgemeinerungen, homogen und spalten-major (zuerst Spalten füllen, danach Zeilen). Zugriff via \texttt{[i,j]}; Lineare Algebra direkt verfügbar (\texttt{t(M)} transponiert, \texttt{M \%*\% N} multipliziert). Vorteil: numerische Effizienz; Nachteil: nur ein Datentyp pro Matrix.

\textbf{Arrays} sind $n$-dimensionale Verallgemeinerungen, homogen wie Matrizen. Zugriff via \texttt{arr[i1,\allowbreak ...,in]}, Subsetting z.\,B.\ \texttt{arr[1:5, , 3]}. Einsatz: Zeitreihen mit mehreren Variablen, 3D-Daten (z.\,B.\ NN-Preprocessing).

\definition{Der \textbf{Data Frame} ist R's Tabellen-Pendant: im Gegensatz zur Matrix \textbf{heterogen} -- Spalten dürfen unterschiedliche Typen haben, müssen aber gleich lang sein. Zugriff via \texttt{\$} (\texttt{df\$Name}) oder Bracket-Notation (\texttt{df[, Name]}).}

Data Frames sind das Arbeitspferd der Statistik in R und erlauben Formel-Syntax wie \texttt{lm(y \textasciitilde\ x1 + x2, data=df)}. Vorteil: Flexibilität, nahtlose Integration mit statistischen Funktionen; Nachteil: langsamer als Matrizen bei rein numerischen Berechnungen. Moderne Alternativen: \texttt{tibble} (strikteres Verhalten) und \texttt{data.table} (optimiert für große Datenmengen).

\textbf{Listen} sind R's generische Vektoren -- Elemente beliebigen Typs (Zahlen, Strings, Listen, Funktionen). Zugriff: \texttt{list[[i]]} (Element) vs.\ \texttt{list[i]} (Sub-Liste); benannte Elemente via \texttt{list\$\allowbreak name}. Einsatz: Funktionsrückgaben, komplexe heterogene Strukturen. Nachteil: langsamer als homogene Strukturen.

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{1.5cm} C{1.0cm} L{4.5cm} L{6.0cm}}
    \textbf{Struktur} & \textbf{Dim} & \textbf{Eigenschaft}     & \textbf{Use-Case}                        \\
    \midrule
    Vektor            & 1D           & Homogen, vektorisiert    & Numerische Sequenzen, Zeitreihen         \\
    Matrix            & 2D           & Homogen, spalten-major   & Lineare Algebra, numerische Berechnungen \\
    Array             & nD           & Homogen, beliebig        & Multidimensionale Daten, Simulationen    \\
    Data Frame        & 2D           & Heterogen (spaltenweise) & Statistische Analyse, Tabellen           \\
    Liste             & nD           & Heterogen, rekursiv      & Komplexe Strukturen, Rückgabewerte       \\
  \end{tabularx}
\end{table}

\subsection{Objektorientierte Programmierung in R}

\textbf{Überblick}

R ist primär funktional -- OOP wurde nachträglich ergänzt, daher existieren mehrere konkurrierende Systeme: \textbf{S3}, \textbf{S4}, \textbf{R5 (Reference Classes)} und \textbf{R6}.

\textbf{S3 -- Informelles Dispatch-System}

Das älteste und meistverwendete System, basiert auf \textbf{generischen Funktionen und Method Dispatch}: eine generische Funktion (\texttt{print}, \texttt{summary}, \texttt{plot}) ruft die passende Methode anhand des \texttt{class}-Attributs des ersten Arguments auf. Bei \texttt{speak(fido)} sucht R nach \texttt{speak.Dog}, dann \texttt{speak.default}; Vererbung läuft über einen \texttt{class}-Vektor mit mehreren Einträgen.

\textbf{Vorteile:} leichtgewichtig, keine formale Definition nötig, kompatibel mit Base-R. \textbf{Nachteile:} keine Klassenvalidierung, fehleranfällig in großen Projekten.

\textbf{S4 -- Formales Klassen-System}

Formales System mit expliziter Klassendefinition, Slot-Validierung und Mehrfachvererbung; verbreitet in Bioconductor.

\begin{itemize}[leftmargin=1.5em,noitemsep]
  \item \textbf{Formale Deklaration} via \texttt{setClass} -- Slots und Typen explizit definiert
  \item \textbf{Slot-Zugriff} via \texttt{@} (nicht \texttt{\$}); \textbf{Validierung} via \texttt{validity}-Argument
  \item \textbf{Mehrfachvererbung} via \texttt{contains}; \textbf{Multiple Dispatch} via \texttt{setMethod}
\end{itemize}
\textbf{Vor-/Nachteile:} formal, sicher, dokumentierbar -- aber verbos und komplexer als S3.

\textbf{R5 / Reference Classes -- Mutable OOP}

Bringt echte \textbf{Referenzsemantik} nach R (wie Python/Java-Klassen): Objekte sind mutable, Methoden ändern Felder in-place. Vorteil bei Zustandsautomaten/Caches; Nachteil: ungewohnt, schwieriger zu debuggen, kaum in Standard-Paketen genutzt.

\textbf{R6 -- Modernes, schlankes Referenz-System}

Moderne Alternative zu R5 mit sauberer Syntax, echter privater/öffentlicher Felddefinition und besserer Performance.

\subsection{Tidyverse -- Philosophie \& Funktionsweise}

\textbf{Was ist das Tidyverse?}

Ein \textbf{kohärentes Ökosystem von R-Paketen} (Hadley Wickham/Posit) mit gemeinsamer Design-Philosophie, konsistenter API und Datenstruktur. Kernpakete: \texttt{ggplot2, dplyr, tidyr, readr, purrr, tibble, stringr, forcats}.

\textbf{Tidy Data}

Kernkonzept (Wickham, 2014): (1) jede \textbf{Variable} eine \textbf{Spalte}, (2) jede \textbf{Beobachtung} eine \textbf{Zeile}, (3) jeder \textbf{Wert} in genau einer Zelle. Tidy Data ist das standardisierte Eingabeformat aller Tidyverse-Funktionen und eliminiert häufiges Reshaping.

\textbf{Beispiel} -- ein typisches ``Wide''-Format mit einer Spalte pro Jahr ist \emph{nicht} tidy, da die Variable ``Jahr'' über mehrere Spalten verteilt ist:
\begin{center}
\small
\begin{tabular}{lcc} \toprule
Land & 2023 & 2024 \\ \midrule
AT & 4{,}1 & 4{,}3 \\
DE & 3{,}8 & 3{,}9 \\ \bottomrule
\end{tabular}
\quad
$\xrightarrow{\texttt{pivot\_longer()}}$
\quad
\begin{tabular}{lcc} \toprule
Land & Jahr & Wert \\ \midrule
AT & 2023 & 4{,}1 \\
AT & 2024 & 4{,}3 \\
DE & 2023 & 3{,}8 \\
DE & 2024 & 3{,}9 \\ \bottomrule
\end{tabular}
\end{center}
Erst die rechte (lange) Form ist tidy: ``Jahr'' und ``Wert'' sind jeweils eigene Spalten, jede Zeile ist genau eine Beobachtung -- und erst in dieser Form lässt sich z.\,B. \texttt{ggplot2} direkt anwenden (\texttt{aes(x = Jahr, y = Wert, color = Land)}).

\textbf{Tibble \& dplyr}

\texttt{tibble} ist ein überarbeiteter Data Frame: besseres Druckverhalten, striktere Subsetting-Semantik, schnellere Erstellung.

\texttt{dplyr} stellt fünf \textbf{Kern-Verben} bereit: \textbf{filter()} (Zeilen filtern), \textbf{select()} (Spalten wählen), \textbf{mutate()} (neue Spalte berechnen), \textbf{arrange()} (sortieren), \textbf{summarise()} (aggregieren). Der \textbf{Pipe-Operator} \texttt{|>} (R\,4.1+) bzw.\ \texttt{\%>\%} (magrittr) übergibt das linke Ergebnis als erstes Argument rechts -- eine lesbare Transformationskette. \texttt{group\_by()} + \texttt{summarise()} ermöglichen Aggregation pro Gruppe. Auf der Long-Tabelle von oben liefert z.\,B.
\begin{center}
\texttt{daten |> filter(Jahr == 2024) |> arrange(desc(Wert))}
\end{center}
in Worten: ``nimm die Daten, filtere auf Jahr 2024, sortiere absteigend nach Wert'' -- jede Zeile der Pipe entspricht einem Verb, lesbar von oben nach unten statt verschachtelt wie \texttt{arrange(filter\allowbreak (daten, Jahr == 2024), desc(Wert))} in Base-R-Stil.

\textbf{tidyr \& ggplot2}

\texttt{tidyr}: \textbf{pivot\_longer()} (Wide$\to$Long, für ggplot2 oft nötig), \textbf{pivot\_wider()} (Long$\to$Wide), \textbf{unnest()} (verschachtelte Daten expandieren).

\texttt{ggplot2} implementiert die \textbf{Grammar of Graphics}: ein Plot entsteht aus Layers mit Daten (\texttt{aes}) und geometrischen Objekten (\texttt{geom}) -- deklarativ statt der manuellen Koordination von \texttt{plot(), points(), lines()} in Base-R.

\textbf{purrr}

Konsistente \texttt{map\_*}-Familie als typsichere Alternative zu \texttt{lapply}/\texttt{sapply}: \texttt{map()} liefert immer eine Liste, \texttt{map\_dbl()} garantiert einen double-Vektor, \texttt{map2()} verarbeitet zwei Listen parallel. Die \texttt{\textasciitilde}-Syntax erzeugt anonyme Funktionen (\texttt{.x} referenziert das aktuelle Element).

\subsection{Wichtige Dateiformate}

Dateiformate lassen sich nach drei Dimensionen klassifizieren: \textbf{Text vs.\ Binär} (CSV/JSON/YAML/XML menschenlesbar vs.\ HDF5/Parquet/Pickle kompakt \& schnell), \textbf{hierarchisch vs.\ tabellarisch} (XML/JSON/YAML beliebig tief vs.\ CSV/Parquet flach optimiert), \textbf{Schema vs.\ schemalos} (XML formal via XSD, JSON optional, CSV/YAML schemalos).

\begin{table}[H]
  \centering
  \begin{tabularx}{\linewidth}{L{1.8cm} L{1.3cm} L{6cm} L{5cm}}
    \textbf{Format} & \textbf{Typ} & \textbf{Stärken} & \textbf{Schwächen} \\
    \midrule
    CSV & Text & universell, einfach, kaum Parsing-Overhead & kein Typ-System, Encoding-Probleme, langsam bei großen Dateien \\
    XML & Text & hierarchisch, formale Validierung (XSD), im Unternehmensumfeld verbreitet & sehr verbose, teuer zu parsen \\
    JSON & Text & kompakt, direkt auf Dicts/Arrays abbildbar, Standard für REST-APIs & keine Kommentare, kein Datumstyp, kein Integer/Float-Unterschied \\
    YAML & Text & maximal lesbar (Einrückung statt Tags), Standard für Configs (Docker, Kubernetes, CI/CD) & Parsing komplex \& fehleranfällig; \textbf{Sicherheit:} immer \texttt{yaml.safe\_load()}, nie \texttt{yaml.load()} aus unbekannten Quellen \\
    HDF5 & Binär & hierarchisch (Groups/Datasets), Lazy Loading, integrierte Kompression, Standard für Keras-Gewichte & nicht lesbar, eingeschränktes Parallelschreiben \\
    Parquet & Binär & spaltenorientiert -- liest nur benötigte Spalten, ideal für OLAP; Standard in Spark/Athena/BigQuery/DuckDB & nur für Tabellendaten, Big-Data-Tooling nötig \\
  \end{tabularx}
\end{table}

Weitere Formate: \textbf{Pickle} -- Pythons native Serialisierung, praktisch aber \textbf{sicherheitskritisch} (nie aus unbekannten Quellen laden). \textbf{NetCDF} -- Standard in Klimawissenschaft/Meteorologie für multidimensionale Array-Daten.

\subsection{git}

\textbf{Was ist Git?}

Git ist ein \textbf{verteiltes Versionskontrollsystem} (DVCS), ursprünglich 2005 von Linus Torvalds geschrieben. Im Gegensatz zu zentralisierten Systemen (SVN, CVS) hat jeder Entwickler eine \textbf{vollständige Kopie} des Repositories -- inkl. kompletter Historie -- auf seinem Rechner.

\textbf{Kernkonzepte}

\textbf{Das Git-Objektmodell:} Git speichert \textbf{Snapshots}, nicht Deltas. Vier Objekttypen im \texttt{.git/objects/}:
\begin{itemize}
  \item \textbf{Blob:} Dateiinhalt (beliebige Binärdaten)
  \item \textbf{Tree:} Verzeichnisstruktur (Liste von Blobs und Sub-Trees)
  \item \textbf{Commit:} Snapshot eines Projektzustands mit Author, Zeitstempel, Parent-Commit
  \item \textbf{Tag:} Benannte Referenz auf einen Commit (z.B. \texttt{v1.0.0})
\end{itemize}

Alle Objekte werden durch SHA-1-Hash (40-stellige Hex-Zeichenkette) identifiziert. Jede Änderung ergibt einen neuen Hash -- Historia ist kryptographisch manipulationssicher.

\textbf{Die drei Bereiche:}
\begin{itemize}
  \item \textbf{Working Directory:} Deine Dateien auf der Festplatte
  \item \textbf{Staging Area:} Zwischenspeicher für \texttt{git add} -- man wählt aus, was in den nächsten Commit geht
  \item \textbf{Repository (.git/):} Die lokale Datenbank mit aller Historie
\end{itemize}

\textbf{Branches und Merging}

Ein \textbf{Branch} ist nur ein Zeiger auf einen Commit. Erstellen ist eine $\mathcal{O}(1)$-Operation -- kein Kopieren von Dateien.

\textbf{Merge:} Zwei Commit-Linien zusammenführen. Zwei Typen:
\begin{itemize}
  \item \textbf{Fast-Forward:} Wenn eine Branch einfach weiter vorne ist, wird der Zeiger verschoben
  \item \textbf{Rebase:} Alternative -- schreibt Commits so um, als wären sie auf aktuellem Stand entstanden. Erzeugt lineare Geschichte, verändert aber SHA-1-Hashes (daher nicht auf shared Branches!)
\end{itemize}

\textbf{Basisbefehle}

\textbf{Repository einrichten:}
\begin{itemize}
  \item \texttt{git init} -- Neues lokales Repo erstellen
  \item \texttt{git clone <url>} -- Bestehendes Repo klonen
\end{itemize}

\textbf{Status und Unterschiede:}
\begin{itemize}
  \item \texttt{git status} -- Zustand anzeigen
  \item \texttt{git diff} -- Änderungen im Working Dir
  \item \texttt{git log --oneline --graph} -- Geschichte mit Branch-Grafik
\end{itemize}

\textbf{Änderungen erfassen:}
\begin{itemize}
  \item \texttt{git add <datei>} -- Datei stagen
  \item \texttt{git add -p} -- Interaktiv: nur bestimmte Hunks
  \item \texttt{git commit -m Nachricht} -- Commit mit Message
  \item \texttt{git commit --amend} -- Letzten Commit korrigieren
\end{itemize}

\textbf{Branches:}
\begin{itemize}
  \item \texttt{git branch} -- Branches auflisten
  \item \texttt{git switch -c <branch>} -- Branch erstellen und wechseln
  \item \texttt{git merge <branch>} -- Mergen
  \item \texttt{git rebase main} -- Rebasing
\end{itemize}

\textbf{Remote-Operationen:}
\begin{itemize}
  \item \texttt{git fetch origin} -- Änderungen vom Remote holen (OHNE merge)
  \item \texttt{git pull origin main} -- fetch + merge
  \item \texttt{git push origin main} -- Commits zum Remote senden
\end{itemize}

\textbf{Rückgängig machen:}
\begin{itemize}
  \item \texttt{git restore <datei>} -- Working-Dir-Änderung verwerfen
  \item \texttt{git revert <hash>} -- Neuen Commit erstellen, der Commit rückgängig macht
  \item \texttt{git reset --hard HEAD\textasciitilde 1} -- Letzten Commit rückgängig (DATEN VERLOREN!)
  \item \texttt{git reflog} -- Rettungsanker nach \texttt{reset --hard}
\end{itemize}

\textbf{Workflow und Best Practices}

\textbf{Typischer Feature Branch Workflow:}
\begin{enumerate}
  \item \texttt{git switch main \&\& git pull origin main}
  \item \texttt{git switch -c feature/my-feature}
  \item Entwickeln und committen
  \item \texttt{git push -u origin feature/my-feature}
  \item Pull Request erstellen für Code Review
  \item Nach Merge: \texttt{git branch -d feature/my-feature}
\end{enumerate}

\textbf{.gitignore:} Definiert, welche Dateien Git ignorieren soll. Typische Einträge:
\begin{itemize}
  \item \texttt{\_\_pycache\_\_/}, \texttt{*.pyc}, \texttt{.venv/} (Python)
  \item \texttt{data/raw/}, \texttt{*.csv}, \texttt{*.h5}, \texttt{*.pkl} (Daten -- nutze Git LFS oder DVC!)
  \item \texttt{.vscode/}, \texttt{.idea/} (IDE)
\end{itemize}
