\section{Modellierung und Simulation}

\subsection{Grundbegriffe: Modell, Modellierung, Simulation}

\definition{Eine \textbf{Hypothese} ist eine allgemeingültige Annahme über Sachverhalte, die eine Beziehung zwischen $\geq 2$ Variablen formuliert. Sie muss als \emph{Konditionalsatz} ("Wenn--dann"/"Je--desto") vorliegen und muss (zumindest potentiell) \textbf{falsifizierbar} sein -- zweiwertig (widerlegbar) oder graduell.}

\definition{Eine \textbf{Theorie} ist eine Vernetzung von gut bewährten Hypothesen. Mit zunehmendem Grad ihrer Bewährung erlangt sie den Charakter einer \emph{Gesetzmäßigkeit}.}

\textbf{Variable vs.\ Parameter:}
\[
\boldsymbol{y} = f(x) + \varepsilon, \quad \text{Beispiel: } y(x) = a \cdot x^2 + \varepsilon
\]
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Variable} ($x, y, \varepsilon$): Symbol für eine Menge von Merkmalsausprägungen
    \begin{itemize}[leftmargin=1.5em]
        \item \textit{Unabhängige Variable} ($x$): gehört zum "Wenn-/Je-Teil" (Ursache/Bedingung)
        \item \textit{Abhängige Variable} ($y$): gehört zum "Dann-/Desto-Teil" (Wirkung/Konsequenz)
        \item \textit{Störvariable} ($\varepsilon$): nicht erfasste Ursache (vergessene unabhängige Variable)
    \end{itemize}
    \item \textbf{Parameter} ($a$): festgesetzter Wert, Teil der Struktur, nicht durch Variablen beeinflusst
\end{itemize}

\definition{Ein \textbf{Modell} ist eine vereinfachte, formale (meist mathematische) Abbildung einer partiellen Realität -- notwendigerweise \emph{Abstraktionen} bzw.\ "Karikaturen" des Realsystems. Maxime: so einfach wie möglich, aber so komplex wie notwendig. Jedes Modell wird für einen \textbf{Modellzweck} entwickelt.}

\definition{Die \textbf{Modellierung} ist der Prozess der Erstellung eines Modells: (1) Nichtformale Beschreibung, (2) Semiformale Beschreibung ("Modellkonzept"), (3) Streng formale mathematische Formulierung.}

\definition{Ein \textbf{Experiment}} ist eine empirische Forschungsmethode, bei der eine oder mehrere unabhängige Variablen variiert werden.

\definition{Eine \textbf{Simulation}} ist im weiteren Sinne der Gesamtkomplex aus Modellieren, Simulieren und Auswerten. Im engeren Sinne: virtuelles Experiment auf Basis eines Simulationsmodells (fast ausschließlich Computersimulation).}

\definition{Ein \textbf{Szenario}} ist eine bestimmte Zusammenstellung von Modellannahmen bzw.\ Konfiguration der unabhängigen Variablen für eine Simulation (vgl.\ "Treatments" bei Experimenten).

\textbf{Typen von Computersimulationen:}
\begin{itemize}[leftmargin=1.5em]
    \item Besseres Verstehen eines \textit{bekannten} Szenarios (Vergangenheit)
    \item Vorhersage von Verhalten in \textit{zukünftigen} Szenarien (z.\,B.\  Klimamodelle)
    \item Optimierung und Entscheidungsunterstützung
\end{itemize}

\subsection{Systeme und Wirkungsgraphen}

\definition{Ein \textbf{System}} ist ein abgegrenzter Teil der Realität, bestehend aus Elementen und deren Wechselwirkungen. \textbf{Dynamische Systeme} sind Systeme, deren Zustandsgrößen sich im Zeitablauf ändern.

\definition{Eine \textbf{Wirkungsbeziehung}} ist ein gerichteter Einfluss einer Systemgröße auf eine andere ($\textit{ceteris paribus}$). \textbf{Positive Wirkung} ($+$): Zunahme der Ursache bewirkt Zunahme der Wirkung. \textbf{Negative Wirkung} ($-$): Zunahme der Ursache bewirkt Abnahme der Wirkung.

\definition{Ein \textbf{Wirkungsgraph}} ist ein gewichteter, gerichteter Graph (Digraph), in dem Systemgrößen die \emph{Knoten} und Wirkungen die \emph{Kanten} (mit Vorzeichen) bilden.

\textbf{Grundregeln nach Bossel (ausgewählt):}
\begin{itemize}[leftmargin=1.5em]
    \item Systemgrößen bilden Knoten, Wirkungen die Kanten
    \item Nur \emph{direkte} Wirkungen dürfen aufgenommen werden
    \item Der Graph gilt nur für einen bestimmten \emph{Ausgangszustand}
\end{itemize}

\textbf{Rückkopplungen:}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Positiver Kreislauf}: Produkt aller Vorzeichen ist positiv $\Rightarrow$ \emph{selbstverstärkend/destabilisierend}
    \item \textbf{Negativer Kreislauf}: Produkt aller Vorzeichen ist negativ $\Rightarrow$ \emph{selbstregulierend/stabilisierend}
\end{itemize}

\definition{Die \textbf{Wirkungsmatrix}} $W$ stellt den Wirkungsgraphen kompakt dar: $w_{i,j}$ gibt Wirkungsintensität von Knoten $j$ auf $i$ an.

\textbf{Graphentheoretische Eigenschaften:}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Schlingen}: Bögen, die einen Knoten mit sich selbst verbinden (direkte Selbstwirkung)
    \item \textbf{Parallele Bögen}: Mehrere gleichgerichtete Bögen zwischen zwei Knoten (zu vermeiden)
    \item \textbf{Schlichtheit}: Graph ohne Schlingen und parallele Bögen
\end{itemize}

\subsection{Eigenschaftsdimensionen von Modellen}

\textbf{Verhaltensbeschreibend vs.\ systemerklärend:}
\begin{itemize}[leftmargin=1.5em]
    \item \textit{Systemerklärend} (glass box): gleiche Wirkungsstruktur und funktionsgleiche Elemente
    \item \textit{Verhaltensbeschreibend} (black box): ähnliches Verhalten, Struktur völlig anders
    \item \textit{Mischform}: grey box / opaque box
\end{itemize}

\textbf{Aggregiert vs.\ individuell:}
\begin{itemize}[leftmargin=1.5em]
    \item \textit{Aggregiert}: statistische Mittelwerte (z.\,B.\  System Dynamics)
    \item \textit{Individuell}: Schlüsselaktöre explizit modelliert (z.\,B.\  agentenbasiert)
\end{itemize}

\textbf{Linear vs.\ nichtlinear:}
\begin{itemize}[leftmargin=1.5em]
    \item \textit{Homogen linear}: $f(\vec{x}) = A \cdot \vec{x}$ (analytisch lösbar)
    \item \textit{Inhomogen linear}: $f(\vec{x}) = A \cdot \vec{x} + \vec{b}$ (gut handhabbar)
    \item \textit{Nichtlinear}: numerisch/simulativ lösbar
\end{itemize}

\textbf{Zeitkontinuierlich vs.\ zeitdiskret} | \textbf{Wertkontinuierlich vs.\ wertdiskret} | \textbf{Raumkontinuierlich vs.\ raumdiskret}

\textbf{Deterministisch vs.\ stochastisch:}
\begin{itemize}[leftmargin=1.5em]
    \item \textit{Deterministisch}: keine Zufälligkeit
    \item \textit{Stochastisch}: Wahrscheinlichkeitsmodellierung (z.\,B.\  Monte-Carlo)
\end{itemize}

\textbf{Wahrscheinlichkeit (Kolmogorov):} $P: \mathcal{A} \to [0,1]$ mit $P(\emptyset)=0$, $P(\Omega)=1$, $P(X \cup Y) = P(X) + P(Y)$ für $X \cap Y = \emptyset$.

\textbf{Ungleichung von Tschebyscheff:}
\[
P(|x - \mu| \geq k\sigma) \leq \frac{1}{k^2}
\]

\textbf{Spezielle Parameter-Eigenschaften:}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Numerisch vs.\ nicht-numerisch}: quantifizierbar vs.\ nicht quantifizierbar
    \item \textbf{Konstant vs.\ zeitvariant}: $a$ vs.\ $a(t)$
    \item \textbf{Realparameter vs.\ Anpassung}: realer vs.\ gefitteter Parameterwert
\end{itemize}

\textit{Hinweis:} Gewöhnliche System-Dynamics-Modelle modellieren räumliche Ausdehnungen oft gar nicht.

\subsection{Pulsmodelle und deren Dynamik}

\definition{Ein \textbf{Pulsmodell}} ermöglicht die diskrete, lineare sowie dynamische Betrachtung der Ausbreitung von Systemstörungen. Ein \textbf{Puls}} ist eine Systemstörung an einem bestimmten Knoten zu einem bestimmten Zeitpunkt. \emph{Hinweis:} Ein Pulsmodell ist \emph{kein} vollständiges Simulationsmodell -- es betrachtet nur die Fortpflanzung von Störungen.

\textbf{Mathematische Formulierung:}
\[
\vec{p}_k = A^k \cdot \vec{p}_0
\]
wobei $A$ die Wirkungsmatrix ist und $\vec{p}_0$ der Startvektor (darf \emph{nicht} aus dem Kern entstammen, insbesondere $\vec{p}_0 \neq \vec{0}$). Kern: $\ker(A) = \{\vec{p} \in \mathbb{R}^n : A\vec{p} = \vec{0}\}$.

\definition{Ein Pulsmodell ist \textbf{pulsstabil}}, wenn alle Eigenwerte im Betrag $\leq 1$ sind: $\forall i: |\lambda_i(A)| \leq 1$. Der \textbf{Spektralradius}} $\rho(A) = \max_i |\lambda_i(A)|$ ist der Radius des kleinsten Kreises, der alle Eigenwerte enthält (Gaußscher Zahlenebene).

\textbf{Eigenkomponenten nach Eigenwertbetrag:}
\begin{itemize}[leftmargin=1.5em]
    \item \textit{Dominante} ($|\lambda_i| = \rho(A)$): bestimmen Langzeitverhalten
    \item \textit{Zunehmende} ($|\lambda_i| > 1$): Pulse wachsen unbegrenzt
    \item \textit{Stationäre} ($|\lambda_i| = 1$): Pulse konstant
    \item \textit{Abnehmende} ($|\lambda_i| < 1$): Pulse klingen ab
    \item \textit{Rotierende/Alterierende}: Oszillationen und Vorzeichenwechsel
\end{itemize}

\textbf{Kontraktion (Spektralnorm):}
\[
\|A\|_S = \sqrt{\rho(A^T A)}, \quad \|\vec{p}_k\|_2 \leq \|A\|_S \cdot \|\vec{p}_{k-1}\|_2
\]
\textit{Achtung:} $\rho(A) \neq \|A\|_S$ im Allgemeinen!

\definition{\textbf{Dualität:}} $\mathrm{spec}(A) = \mathrm{spec}(A^T)$ $\Rightarrow$ Pulsmodell und duales Modell (umgekehrte Wirkungsrichtungen) haben gleiche Pulsdynamik.

\textbf{Zusammenhang Spur, Determinante und Eigenwerte:}
\[
\mathrm{Spur}(A) = \sum_i \lambda_i, \quad \det(A) = \prod_i \lambda_i
\]

\subsection{Diskrete Markow-Ketten}

\definition{Eine \textbf{diskrete Markow-Kette 1.\ Ordnung}} ist ein stochastischer Prozess, bei dem der \emph{zukünftige Zustand nur vom aktuellen Zustand}} abhängt (Markow-Eigenschaft / Gedächtnislosigkeit). Kanten-Gewichte geben \emph{Übergangswahrscheinlichkeiten} an.

\definition{Die \textbf{Übergangsmatrix}} (stochastische Matrix) $P$ erfüllt: $\forall i,j: p_{i,j} \geq 0$ und $\forall j: \sum_i p_{i,j} = 1$. Jede \emph{Spalte}} summiert sich zu 1.

\textbf{Zeitliche Entwicklung:}
\[
\vec{g}^{(k)} = P^k \cdot \vec{g}^{(0)}
\]
wobei $\vec{g}^{(0)}$ die Startverteilung ist ($\sum_i g_i = 1$).

\definition{Die \textbf{stationäre Verteilung}} ist der normierte Eigenvektor zum Eigenwert $\lambda = 1$:
\[
P \cdot \vec{\zeta} = \vec{\zeta} \quad \Rightarrow \quad \lim_{k \to \infty} P^k \cdot \vec{g}^{(0)} = \vec{\zeta}
\]

\textbf{Sonderfälle:}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Nicht zusammenhängender Graph}: Eigenwert 1 tritt mehrfach auf (je Komponente einmal)
    \item \textbf{Alternierender Graph}: Neben $+1$ tritt auch $-1$ als Eigenwert auf
\end{itemize}

\textbf{Berechnung der stationären Verteilung:} (1) Eigenwerte von $P$, (2) Eigenvektor zu $\lambda = 1$, (3) Normalisierung: $\vec{\zeta}_{\mathrm{norm}} = \frac{\vec{\zeta}}{\sum_i \zeta_i}$.

\textbf{Unterschied zu Pulsmodellen:} Bei stochastischen Matrizen sind die \emph{Spaltensummen}} gleich 1 (normiert), bei Pulsmodellen nicht. Iterationsvorschrift ist formal identisch.

\subsection{System Dynamics}

\definition{\textbf{System Dynamics (SD)}} ist eine Methode zur Modellierung und Simulation \emph{dynamischer, rückgekoppelter Systeme}} mittels Differentialgleichungen. Im Zentrum stehen \emph{Zustandsgrößen}} (Bestandsgrößen), die durch Flussgrößen verändert werden. SD modelliert typischerweise aggregierte, zeitkontinuierliche Systeme.

\textbf{Arten von Systemgrößen in SD:}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Zustandsgrößen} (Stocks): Systemzustand zu jedem Zeitpunkt; benötigen Anfangswert (Darstellung: Rechteck)
    \item \textbf{Zustandsveränderungsgrößen} (Flows): Zu- oder Abflüsse von Zustandsgrößen (Darstellung: Ventilpfeil)
    \item \textbf{Zwischen-/Hilfsgrößen}: Direkt aus anderen Größen berechenbar
    \item \textbf{Vorgabe-/Eingriffsparameter}: Konstante oder zeitvariante, exogen definierte Größen
\end{itemize}

\textbf{Grundstruktur:}
\[
\frac{d(\text{Zustand})}{dt} = \text{Zufluss} - \text{Abfluss}
\]

\beispiel{Bevölkerungsmodell: $\frac{d(\mathrm{Bev})}{dt} = \mathrm{Geburtenrate} \cdot \mathrm{Bev} - \mathrm{Sterberate} \cdot \mathrm{Bev}$}

\definition{Die \textbf{Euler-Cauchy-Integration}} ist die numerische Integrationsmethode bei SD-Simulation:
\[
z(t + \Delta t) \approx z(t) + \dot{z}(t) \cdot \Delta t
\]

\textbf{Unterschied SD vs.\ allgemeine Systemtheorie:} SD modelliert keine explizite Systemumgebung -- alle Einflüsse werden als Teil des Modells behandelt. Trajektorien in autonomen SD-Modellen können sich niemals schneiden.

\subsection{Zustandsräume, Zustandsvektoren, Trajektorien und Attraktoren}

\definition{Die \textbf{Dimensionalität}} eines Systems ist die Anzahl der Zustandsgrößen. Die Wahl der Zustandsgrößen ist nicht eindeutig, die Dimensionalität hingegen schon.

\definition{Der \textbf{Zustandsraum}} (Phasenraum) wird durch die Zustandsgrößen aufgespannt. \emph{Keine Zeitachse!} Die Zeit fließt implizit durch die Veränderung des Zustandsvektors ein.

\definition{Der \textbf{Zustandsvektor}} $\vec{z}(\tau)$ beschreibt einen Systemzustand im Zustandsraum. Eine \textbf{Trajektorie}} ist die Kurve im Zustandsraum, die den Weg des Zustandsvektors beschreibt ("Geschichte des Systems").

\definition{Die \textbf{Zustandsebene}} wird von zwei Zustandsgrößen aufgespannt. Ein \textbf{Zustandsbild}} ist die Projektion der Zustandsbahn in einen niederdimensionaleren Raum.

\textbf{Grundtypen von Attraktoren:}
\begin{itemize}[leftmargin=1.5em]
    \item \textbf{Punktattraktor}: Zustandsbahnen konvergieren auf stabilen Gleichgewichtspunkt
    \item \textbf{Grenzzyklus}: Zustandsbahnen laufen auf geschlossene Kurven zu
    \item \textbf{Torus}: Zyklus höherer Ordnung (bei $\geq 3$ Zustandsgrößen)
    \item \textbf{Seltsamer Attraktor}: Chaotisch und/oder fraktal (starke Sensitivität)
\end{itemize}

\definition{Ein \textbf{Repellor}} ist ein instabiler Gleichgewichtspunkt -- Trajektorien entfernen sich von ihm.

\definition{\textbf{Separatrizen}} trennen Regionen unterschiedlichen Verhaltens und verlaufen durch Repelloren (stabile und instabile Mannigfaltigkeit). \textbf{Bifurkationen}} treten auf, wenn Parameteränderung qualitativ anderes Verhalten ergibt.

\textit{Hinweis (autonome Systeme):} Trajektorien können sich niemals schneiden, da der Systemzustand die zukünftige Dynamik eindeutig bestimmt.

\subsection{Dynamische lineare Systeme}

\definition{Ein \textbf{dynamisches System}} wird formal beschrieben durch die \textbf{Zustandsgleichung}} und \textbf{Verhaltensgleichung}:
\[
\dot{z}(t) = f(z(t), u(t), t), \quad v(t) = g(z(t), u(t), t)
\]

\textbf{Diskrete und autonome Varianten:}
\begin{itemize}[leftmargin=1.5em]
    \item \textit{Kontinuierlich, nichtautonom}: $\dot{z}(t) = f(z(t), u(t), t)$
    \item \textit{Kontinuierlich, autonom}: $\dot{z}(t) = f(z(t))$
    \item \textit{Diskret}: $z_{k+1} = f(z_k, u_k, k)$ oder $z_{k+1} = f(z_k)$
\end{itemize}

\definition{Die \textbf{Linearisierung}} nichtlinearer Systeme erfolgt mittels Jacobi-Matrizen am Arbeitspunkt $(z_0, u_0)$:
\[
\Delta\dot{z}(t) = A \cdot \Delta z(t) + B \cdot \Delta u(t)
\]

\definition{\textbf{LTI-Systeme}} (Linear Time-Invariant) haben konstante $A$ und $B$. Lösung:
\[
z(t) = e^{At} \cdot z_0 + \int_0^t e^{A(t-\tau)} B\, u(\tau)\, d\tau
\]

\textbf{Stabilität basierend auf Eigenwerten von} $A$:
\begin{itemize}[leftmargin=1.5em]
    \item $\mathrm{Re}(\lambda_i) < 0$ für alle $i$: \textbf{asymptotisch stabil} (Trajektorien $\to 0$)
    \item $\mathrm{Re}(\lambda_i) > 0$ für mind.\  ein $i$: \textbf{instabil}
    \item $\mathrm{Re}(\lambda_i) = 0$: \textbf{grenzstabil} (Schwingungen)
\end{itemize}

\textit{Hinweis:} LTI-Theorie ist grundlegend für Regelungstechnik und lineare Pulsmodelle. SD modelliert keine explizite Systemumgebung.
